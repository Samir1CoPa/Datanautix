Calendar = ADDCOLUMNS (
 CALENDAR (date(Year(today())-1,1,1),date(year(today()), 12, 31)),
 "DateAsInteger", FORMAT ( [Date], "YYYYMMDD" ),
 "ShortDate", Format ([Date], "MM/DD/YYYY"),
 "Year", YEAR ( [Date] ),
 "Month", FORMAT ( [Date], "MM" ),
 "Weeknumber", weeknum([Date],1),
 "YearMonthnumber", FORMAT ( [Date], "YYYY/MM" ),
 "YearWeeknumber", FORMAT ( [Date], "YYYY"&format(weeknum([Date],1), "00") ),
 "YearWeekDate", weeknum([Date],1)&FORMAT ( [Date], " YYYY"),
 "Week", FORMAT ( [date]+1-weekday([Date]), "MM/DD/YY - " ) & FORMAT ( [date]+7-weekday([Date]), "MM/DD/YY"),
 "YearMonthShort", FORMAT ( [Date], "mmm YYYY" ),
 "Month Short Name", FORMAT ( [Date], "mmm" ),
 "Month Name", FORMAT ( [Date], "mmmm" ),
 "DayOfWeekNumber", WEEKDAY ( [Date] ),
 "DayOfWeek", FORMAT ( [Date], "dddd" ),
 "DayOfWeekShort", FORMAT ( [Date], "ddd" ),
 "Quarter", "Q" & FORMAT ( [Date], "Q" ),
 "Begin Shift", FORMAT ("6:00 AM", "hh:nn AM/PM"),
 "End Shift", FORMAT ("2:30 PM", "hh:nn AM/PM"),
 "YearQuarter", "Q" & FORMAT ( [Date], "Q" ) & FORMAT ( [Date], " YYYY" ),
 "Days Outstanding", VALUE(DATEDIFF([Date],TODAY(),DAY)),
"MTD", SWITCH(
TRUE(),
MONTH([Date]) = MONTH(TODAY()) && DAY([Date]) <= DAY(TODAY()), "MTD"),
"YTD", SWITCH(
TRUE(),
MONTH([Date]) < MONTH(TODAY()) || (MONTH([Date]) = MONTH(TODAY()) && DAY([Date]) <= DAY(TODAY())), "YTD"),
"Aging Bucket", SWITCH(
 TRUE(),
 VALUE(DATEDIFF([Date],TODAY(),DAY)) <=0, "Current",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >0 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=30, "1 - 30 Days",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >30 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=60, "31 - 60 Days",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >60 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=90, "61 - 90 Days",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >90 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=120, "90 - 120 Days",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >120 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=150, "120 - 150 Days",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >150, "150+ Days"),
"Aging Bucket Sort", SWITCH(
 TRUE(),
 VALUE(DATEDIFF([Date],TODAY(),DAY)) <=0, "1",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >0 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=30, "2",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >30 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=60, "3",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >60 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=90, "4",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >90 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=120, "5",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >120 && VALUE(DATEDIFF([Date],TODAY(),DAY)) <=150, "6",
 VALUE(DATEDIFF([Date],TODAY(),DAY)) >150, "7"))
